#  Format:  name = value
#  Lines starting with # are notes and are ignored.

#  Yes/no values: yes or no.

# ---------- Shops ----------

# Gold Gabriel (and every other NPC shop) pays when you sell items
# to him. This one is a multiplier, not a percent:
# 1 = normal, 2 = double, 0.5 = half.
sell_multiplier = 1

# Trader Discount: percent off the gold price of everything in the
# NPC shops, Ponty and Lost and Found. 0 = normal price,
# 50 = half price, 100 = free. Fire Sale makes everything free anyway.
trader_discount = 0


# ---------- Ponty (second-hand shop in town) ----------

# Ponty throws out his stock and restocks with random high-tier
# items every this many minutes. 0 = off (normal Ponty).
ponty_restock_minutes = 5

# How many random items Ponty puts on sale each restock.
ponty_item_count = 20

# Lowest item grade Ponty will stock, judged at the level he
# stocks it: 0 = anything, 1 = high grade (the game's "high"
# tier), 2 = rare, 3 = legendary.
ponty_min_grade = 1

# Level range of the gear Ponty stocks.
# Upgradeable gear (weapons, armor) is upgraded to a random level
# between these two numbers.
ponty_upgrade_level_min = 6
ponty_upgrade_level_max = 9
# Compoundable gear (rings, earrings, amulets, ...).
ponty_compound_level_min = 2
ponty_compound_level_max = 4


# ---------- Experience and combat ----------

# Experience from monsters. A multiplier: 1 = normal,
# 2 = double XP, 0.5 = half.
xp_rate = 1

# Damage your characters deal to monsters. A multiplier: 1 = normal,
# 2 = every hit does double damage, 0.5 = half.
damage_rate = 1

# Damage your characters take. A multiplier: 1 = normal,
# 0.5 = you lose half as much HP, 0 = you never lose HP from attacks.
hp_loss = 1

# MP spent on attacks and skills. A multiplier: 1 = normal,
# 0.5 = everything costs half MP, 0 = attacks and skills cost no MP.
mp_usage = 1

# Wait before a skill can be used again. A multiplier: 1 = normal,
# 0.5 = half the wait, 0 = no wait at all.
skill_cooldown = 1

# XP lost when a character dies. A multiplier: 1 = normal,
# 0.5 = half as much, 0 = nothing is lost.
death_penalty = 1


# ---------- Upgrading and compounding ----------

# Multiplies the success chance. 1 = normal, 2 = twice as
# likely, 10 = almost never fails. The chance shown in the
# upgrade/compound window already includes this.
upgrade_success = 1
compound_success = 1

# Safe upgrades: a failed upgrade or compound doesn't destroy
# anything. The item stays at its old level (only the scroll is used).
safe_upgrades = no


# ---------- Drops ----------

# Item drop chance from every monster, everywhere. A multiplier:
# 1 = normal, 2 = every item is twice as likely to drop,
# 0.5 = half as likely, 0 = nothing drops.
drop_rate = 1

# Gold dropped by monsters. A multiplier: 1 = normal, 2 = double.
gold_drop = 1

# Every monster drops every item on its drop list on every kill.
# Very silly. Overrides drop_rate.
drop_everything = no

# Exchange luck: every exchange (Xyn, boxes, gems, quest items),
# fishing, mining and the chests monsters drop. A multiplier:
# 1 = normal, 2 = every reward rarer than the table's most common
# one is twice as likely, 0.5 = half as likely.
exchange_luck = 1


# ---------- Daisy (monster hunts) ----------

# Monster tokens Daisy gives for each finished monster hunt.
# 0 = none.
monster_hunt_tokens = 1


# ---------- Monster respawns ----------

# Time before a killed monster comes back, bosses and event
# monsters included. A multiplier: 1 = normal, 0.5 = twice as
# fast, 0 = straight away.
monster_respawn = 1


# ---------- Cheats ----------

# Fire Sale: everything in the NPC shops, Ponty and Lost and
# Found costs nothing.
fire_sale = no

# God Mode: your characters take no damage at all.
god_mode = no


# ---------- Holiday events ----------
# You can also switch these while playing in the Mod Menu, and
# your choices are saved here. The holidays control their drops
# and event icons; the event monsters are switched separately.
# yes = always on. They were designed to run one at a time, so if
# something looks odd, turn some of them off.

holiday_season = yes
lunar_new_year = yes
valentines = yes
halloween = yes
egg_hunt = yes

# Event monsters, each switched on its own (they spawn even when
# their holiday is off). yes = spawns, no = never spawns.
monster_grinch = yes
monster_snowman = yes
monster_mrpumpkin = yes
monster_mrgreen = yes
monster_slenderman = yes
monster_dragold = yes
monster_pinkgoo = yes
monster_wabbit = yes


# ---------- Multiple characters ----------

# How many of your characters can be logged in at the same time.
max_characters_online = 10

# How fast your characters run. A multiplier: 1 = normal,
# 1.5 = half as fast again, 2 = double. 5 at most.
run_speed = 1


# ---------- Fishing and mining ----------

# Wait between catches/finds. no = fish and mine again straight
# away (each attempt still takes a few seconds).
fishing_mining_cooldown = no

# Can fishing rods and pickaxes break while you use them?
tools_break = no


# ---------- New characters ----------

# Gold every new character starts with.
starting_gold = 10000000

# The amount the Mod Menu's Give Gold sends to every character
# online when you press Send Gold. Kept here so it's remembered.
give_gold = 1000000


# ---------- Merrit (market square visitor) ----------

# Merrit brings a parcel to each merchant's open stall on the
# market square every this many minutes (each merchant has their
# own timer). Normal game: 60, shared by the whole account.
merrit_visit_minutes = 60

# On each visit Merrit buys everything for sale on the stall at
# the price you set. Only the stall is emptied, never the bag.
merrit_buys_stall = yes
